cmake_minimum_required(VERSION 3.15)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(FemusCutFemNative LANGUAGES CXX)
  enable_testing()
endif()
get_filename_component(FEMUS_NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_path(FEMUS_NATIVE_BOOST boost/math/quadrature/gauss_kronrod.hpp HINTS ENV BOOST_INCLUDEDIR)
find_path(FEMUS_NATIVE_EIGEN Eigen/Dense HINTS ENV EIGEN3_INCLUDE_DIR PATH_SUFFIXES eigen3)
if(NOT FEMUS_NATIVE_BOOST OR NOT FEMUS_NATIVE_EIGEN)
  message(FATAL_ERROR "Native CutFEM tests require Boost and Eigen headers")
endif()
file(GLOB FEMUS_NATIVE_GAUSS CONFIGURE_DEPENDS "${FEMUS_NATIVE_ROOT}/src/quadrature/*Points.cpp")
set(FEMUS_NATIVE_FE Line Quadrilateral Triangle Hexaedron Tetrahedral)
set(FEMUS_NATIVE_SOURCES ${FEMUS_NATIVE_GAUSS})
foreach(name IN LISTS FEMUS_NATIVE_FE)
  list(APPEND FEMUS_NATIVE_SOURCES "${FEMUS_NATIVE_ROOT}/src/fe/${name}.cpp")
endforeach()
add_library(cutfem_native_support STATIC ${FEMUS_NATIVE_SOURCES})
target_compile_features(cutfem_native_support PUBLIC cxx_std_14)
target_include_directories(cutfem_native_support PUBLIC "${FEMUS_NATIVE_ROOT}/src/quadrature/cutfem_ls" "${FEMUS_NATIVE_ROOT}/src/quadrature" "${FEMUS_NATIVE_ROOT}/src/enums" "${FEMUS_NATIVE_ROOT}/src/fe" "${FEMUS_NATIVE_BOOST}" "${FEMUS_NATIVE_EIGEN}")
set(FEMUS_NATIVE_TESTS line_integration test_reference_coefficients test_contract direct_parent_moments tri_endpoint_regressions endpoint_surface_regressions endpoint_exponents endpoint_nested_contacts quad_tri_standard_moments tet_standard_moments quad_native_quadrants quad_scaled_plane test_tet_slice_regression test_root_candidate_regression mesh_refinement line_mesh_refinement hex_sphere_boundary_tangencies endpoint_coordinate_resolution hex_curved_endpoint_regressions quad_curved_endpoint_regressions outer_interval_contract review_regressions single_graph_root)
foreach(name IN LISTS FEMUS_NATIVE_TESTS)
  add_executable(cutfem_${name} "${name}.cpp")
  target_link_libraries(cutfem_${name} PRIVATE cutfem_native_support)
  add_test(NAME cutfem_${name} COMMAND cutfem_${name})
endforeach()
add_executable(cutfem_hex_tet_validation hex_tet_validation.cpp)
target_link_libraries(cutfem_hex_tet_validation PRIVATE cutfem_native_support)
add_test(NAME cutfem_hex_sphere_mesh COMMAND cutfem_hex_tet_validation mesh HEX 2 4 1e-9)
add_test(NAME cutfem_tet_sphere_mesh COMMAND cutfem_hex_tet_validation mesh TET 2 4 1e-9)

add_executable(cutfem_outer_interval_benchmark outer_interval_benchmark.cpp)
target_link_libraries(cutfem_outer_interval_benchmark PRIVATE cutfem_native_support)
