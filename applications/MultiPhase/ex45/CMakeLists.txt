CMAKE_MINIMUM_REQUIRED(VERSION 3.15)

get_filename_component(APP_FOLDER_NAME ${CMAKE_CURRENT_LIST_DIR} NAME)
set(THIS_APPLICATION ${APP_FOLDER_NAME})
PROJECT("${APP_FOLDER_NAME_PARENT}_${THIS_APPLICATION}")

set(MAIN_FILE "${THIS_APPLICATION}")
set(EXEC_FILE "${APP_FOLDER_NAME_PARENT}_${MAIN_FILE}")
femusMacroBuildApplication(${MAIN_FILE} ${EXEC_FILE})
target_sources(${EXEC_FILE} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/AlgoimQuadrature.cpp")
target_compile_features(${EXEC_FILE} PRIVATE cxx_std_17)
set_target_properties(${EXEC_FILE} PROPERTIES CXX_EXTENSIONS OFF)

if(NOT TARGET Eigen3::Eigen)
  find_package(Eigen3 REQUIRED NO_MODULE)
endif()
find_path(FEMUS_ALGOIM_LAPACKE_INCLUDE_DIRECTORY NAMES lapacke.h mkl_lapacke.h)
find_library(FEMUS_ALGOIM_LAPACKE_LIBRARY NAMES lapacke mkl_rt)
if(NOT FEMUS_ALGOIM_LAPACKE_INCLUDE_DIRECTORY OR NOT FEMUS_ALGOIM_LAPACKE_LIBRARY)
  message(FATAL_ERROR "ex45 needs LAPACKE headers and a LAPACKE library; install liblapacke-dev or set FEMUS_ALGOIM_LAPACKE_INCLUDE_DIRECTORY and FEMUS_ALGOIM_LAPACKE_LIBRARY")
endif()
target_include_directories(${EXEC_FILE} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../src/quadrature/cutfem" "${FEMUS_ALGOIM_LAPACKE_INCLUDE_DIRECTORY}")
target_link_libraries(${EXEC_FILE} Eigen3::Eigen "${FEMUS_ALGOIM_LAPACKE_LIBRARY}" ${FEMUS_ALGOIM_EXTRA_LIBRARIES})

find_package(OpenMP QUIET)
if(OpenMP_CXX_FOUND)
  target_link_libraries(${EXEC_FILE} OpenMP::OpenMP_CXX)
  target_compile_definitions(${EXEC_FILE} PRIVATE FEMUS_USE_OPENMP=1)
endif()

find_path(FEMUS_EX45_GPERFTOOLS_INCLUDE_DIRECTORY NAMES gperftools/profiler.h)
find_library(FEMUS_EX45_GPERFTOOLS_LIBRARY NAMES profiler)
if(FEMUS_EX45_GPERFTOOLS_INCLUDE_DIRECTORY AND FEMUS_EX45_GPERFTOOLS_LIBRARY)
  target_include_directories(${EXEC_FILE} PRIVATE "${FEMUS_EX45_GPERFTOOLS_INCLUDE_DIRECTORY}")
  target_link_libraries(${EXEC_FILE} "${FEMUS_EX45_GPERFTOOLS_LIBRARY}")
  target_compile_definitions(${EXEC_FILE} PRIVATE FEMUS_EX45_HAS_GPERFTOOLS=1)
endif()

set(FEMUS_EX45_INPUT_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}/../ex44/input")
if(NOT IS_DIRECTORY "${FEMUS_EX45_INPUT_DIRECTORY}")
  message(FATAL_ERROR "ex45 requires the existing ex44/input runtime assets")
endif()
file(GLOB FEMUS_EX45_INPUT_ASSETS RELATIVE "${FEMUS_EX45_INPUT_DIRECTORY}" "${FEMUS_EX45_INPUT_DIRECTORY}/*")
foreach(FEMUS_EX45_INPUT_ASSET IN LISTS FEMUS_EX45_INPUT_ASSETS)
  if(FEMUS_EX45_INPUT_ASSET STREQUAL "FS_solver.options")
    configure_file("${FEMUS_EX45_INPUT_DIRECTORY}/${FEMUS_EX45_INPUT_ASSET}" "${PROJECT_BINARY_DIR}/input/${FEMUS_EX45_INPUT_ASSET}" COPYONLY)
  elseif(NOT EXISTS "${PROJECT_BINARY_DIR}/input/${FEMUS_EX45_INPUT_ASSET}" AND NOT IS_SYMLINK "${PROJECT_BINARY_DIR}/input/${FEMUS_EX45_INPUT_ASSET}")
    file(CREATE_LINK "${FEMUS_EX45_INPUT_DIRECTORY}/${FEMUS_EX45_INPUT_ASSET}" "${PROJECT_BINARY_DIR}/input/${FEMUS_EX45_INPUT_ASSET}" SYMBOLIC RESULT FEMUS_EX45_INPUT_LINK_RESULT)
    if(NOT FEMUS_EX45_INPUT_LINK_RESULT STREQUAL "0")
      message(FATAL_ERROR "Cannot link ex45 runtime asset ${FEMUS_EX45_INPUT_ASSET}: ${FEMUS_EX45_INPUT_LINK_RESULT}")
    endif()
  endif()
endforeach()

if(NOT EXISTS "${PROJECT_BINARY_DIR}/run_ex45.sh" AND NOT IS_SYMLINK "${PROJECT_BINARY_DIR}/run_ex45.sh")
  file(CREATE_LINK "${PROJECT_SOURCE_DIR}/run_ex45.sh" "${PROJECT_BINARY_DIR}/run_ex45.sh" SYMBOLIC RESULT FEMUS_EX45_RUN_LINK_RESULT)
  if(NOT FEMUS_EX45_RUN_LINK_RESULT STREQUAL "0")
    message(FATAL_ERROR "Cannot link ex45 launcher: ${FEMUS_EX45_RUN_LINK_RESULT}")
  endif()
endif()
